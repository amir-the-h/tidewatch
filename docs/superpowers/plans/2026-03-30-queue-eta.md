# Queue ETA Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Show per-action queue ETAs and an accurate overall drain ETA on the Queue page, driven by real execution throughput broadcast live via WebSocket.

**Architecture:** `worker.Manager` maintains a per-action `*rateTracker` (already implemented in `rate.go`) in a `sync.Map`. On each task completion `recordCompletion(actionID)` is called; `throughputPerAction()` derives items/sec from the ring buffer. This is included in `PoolStats`, broadcast via `worker_stats` WS events, and returned in `GET /api/queue`. The frontend computes `ETA = pending / throughput` per action and replaces the existing rolling-sample approach for the overall ETA.

**Tech Stack:** Go (`sync.Map`, existing `rateTracker`), TypeScript/Vue 3, Pinia, Naive UI

```go
// recordCompletion appends a completion timestamp for actionID.
// Called after every task completes (success or error).
func (m *Manager) recordCompletion(actionID int64) {
	v, _ := m.completionTrackers.LoadOrStore(actionID, &rateTracker{})
	v.(*rateTracker).record()
}

// throughputPerAction returns current estimated throughput (tasks/sec) per action ID.
// Actions with fewer than 2 completions or a stale window return 0.
func (m *Manager) throughputPerAction() map[int64]float64 {
	result := make(map[int64]float64)
	m.completionTrackers.Range(func(k, v any) bool {
		result[k.(int64)] = v.(*rateTracker).rate()
		return true
	})
	return result
}
```

```go
// Stats returns a snapshot of worker pool activity.
func (m *Manager) Stats() PoolStats {
	active := int(m.active.Load())
	idle := m.totalWorkers - active
	if idle < 0 {
		idle = 0
	}

	var activeWorkers []WorkerInfo
	m.workerActions.Range(func(_, v any) bool {
		activeWorkers = append(activeWorkers, v.(WorkerInfo))
		return true
	})

	return PoolStats{
		Total:            m.totalWorkers,
		Active:           active,
		Idle:             idle,
		ActiveWorkers:    activeWorkers,
		ActionThroughput: m.throughputPerAction(),
	}
}
```

```typescript
// web/src/stores/queues.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export interface QueueEntry {
  action_id: number
  action_name: string
  pipeline_name: string
  pipeline_type: string
  pending: number
  workers: Array<{ id: number; backend: string }>
  throughput: number  // items/sec; 0 = not yet established
}

export interface WorkerStats {
  total: number
  active: number
  idle: number
  active_workers: Array<{ id: number; action: string; pipeline: string; backend: string }>
  action_throughput?: Record<number, number>
}

export const useQueuesStore = defineStore('queues', () => {
  const queues = ref<QueueEntry[]>([])
  const workerStats = ref<WorkerStats | null>(null)

  // Set the full list from the initial HTTP fetch (onMounted in QueueView).
  function setQueues(entries: any[]) {
    queues.value = entries.map(e => ({
      ...e,
      workers: (e.workers ?? []).map((w: any) => ({ id: w.id, backend: w.backend ?? '' })),
      throughput: e.throughput ?? 0,
    }))
  }

  // Update a single action's pending count from a WS queue_update event.
  function setQueueEntry(payload: { action_id: number; pending: number }) {
    const entry = queues.value.find(q => q.action_id === payload.action_id)
    if (entry) {
      entry.pending = payload.pending
    }
  }

  // Update pool-level stats and cross-update the workers + throughput fields on
  // each queue entry so both columns stay live without a full page refresh.
  function setWorkerStats(stats: WorkerStats) {
    workerStats.value = stats
    const activeWorkers = stats.active_workers ?? []
    const throughputMap = stats.action_throughput ?? {}
    for (const entry of queues.value) {
      entry.workers = activeWorkers
        .filter(w => w.action === entry.action_name)
        .map(w => ({ id: w.id, backend: w.backend ?? '' }))
      // Only update throughput if the WS event includes the map (avoids zeroing on old events).
      if (stats.action_throughput !== undefined) {
        entry.throughput = throughputMap[entry.action_id] ?? 0
      }
    }
  }

  const totalPending = computed(() =>
    queues.value.reduce((sum, q) => sum + q.pending, 0)
  )

  return { queues, workerStats, totalPending, setQueues, setQueueEntry, setWorkerStats }
})
```

```typescript
function fmtEta(secs: number): string {
  if (secs < 60) return `~${secs}s`
  const m = Math.floor(secs / 60), s = secs % 60
  if (m < 60) return s > 0 ? `~${m}m ${s}s` : `~${m}m`
  const hr = Math.floor(m / 60), rm = m % 60
  return rm > 0 ? `~${hr}h ${rm}m` : `~${hr}h`
}
```

3. **Replace** the existing `drainEta` computed with:

```typescript
const drainEta = computed<string | null>(() => {
  const totalThroughput = queuesStore.queues.reduce((s, q) => s + q.throughput, 0)
  if (totalThroughput === 0 || queuesStore.totalPending === 0) return null
  return fmtEta(Math.ceil(queuesStore.totalPending / totalThroughput))
})
```

```typescript
function perActionEta(entry: QueueEntry): string | null {
  if (entry.throughput === 0 || entry.pending === 0) return null

}
```
