> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Vue 3 SPA in `web/` that exposes every iran-watcher REST API capability through a dark-themed sidebar UI, served by nginx in a separate Docker container.

**Architecture:** Vite + Vue 3 (`<script setup>`) + Naive UI (dark theme) + Pinia (one store per resource) + Vue Router 4. A typed `apiFetch` wrapper in `src/api/client.ts` is the only thing that calls `fetch`. All components read from Pinia stores and call store actions — never the API directly.

**Tech Stack:** Vue 3, Vite, TypeScript, Naive UI, Pinia, Vue Router 4, nginx:alpine, node:22-alpine

**Spec:** `docs/superpowers/specs/2026-03-23-vue-ui-design.md`

// web/src/api/client.ts
const base = (import.meta.env.VITE_API_BASE_URL as string) ?? ''

export class ApiError extends Error {
  constructor(public status: number, message: string) {
    super(message)
    this.name = 'ApiError'
  }
}

export async function apiFetch<T>(path: string, options?: RequestInit): Promise<T> {
  const res = await fetch(`${base}${path}`, {
    headers: { 'Content-Type': 'application/json', ...options?.headers },
    ...options,
  })
  if (!res.ok) {
    const body = await res.json().catch(() => ({ error: res.statusText }))
    throw new ApiError(res.status, (body as any).error ?? res.statusText)
  }
  if (res.status === 204) return undefined as T
  return res.json() as Promise<T>
}
```

// web/src/stores/stream.ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import { getHealth } from '../api/health'
import { getStreamStatus, connectStream, disconnectStream } from '../api/stream'

export const useStreamStore = defineStore('stream', () => {
  const connected = ref(false)
  const healthy   = ref(false)
  const loading   = ref(false)
  const error     = ref<string | null>(null)

  async function fetchStatus() {
    loading.value = true; error.value = null
    try {
      const [health, status] = await Promise.all([getHealth(), getStreamStatus()])
      healthy.value   = health.status === 'ok'
      connected.value = status.connected
    } catch (e: any) { error.value = e.message }
    finally { loading.value = false }
  }

  async function connect() {
    loading.value = true; error.value = null
    try { await connectStream(); connected.value = true }
    catch (e: any) { error.value = e.message; throw e }
    finally { loading.value = false }
  }

  async function disconnect() {
    loading.value = true; error.value = null
    try { await disconnectStream(); connected.value = false }
    catch (e: any) { error.value = e.message; throw e }
    finally { loading.value = false }
  }

  return { connected, healthy, loading, error, fetchStatus, connect, disconnect }
})
```

// web/src/stores/sources.ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import * as api from '../api/sources'
import type { Source, CreateSourceRequest } from '../api/sources'

export const useSourcesStore = defineStore('sources', () => {
  const sources = ref<Source[]>([])
  const loading = ref(false)
  const error   = ref<string | null>(null)

  async function fetchSources() {
    loading.value = true; error.value = null
    try { sources.value = await api.listSources() }
    catch (e: any) { error.value = e.message }
    finally { loading.value = false }
  }

  async function createSource(body: CreateSourceRequest) {
    const src = await api.createSource(body)
    sources.value.push(src)
    return src
  }

  async function deactivate(id: number) {
    await api.deactivateSource(id)
    const s = sources.value.find(s => s.id === id)
    if (s) s.active = false
  }

  async function remove(id: number) {
    await api.deleteSource(id)
    sources.value = sources.value.filter(s => s.id !== id)
  }

  return { sources, loading, error, fetchSources, createSource, deactivate, remove }
})

    const r = await api.updateRule(id, body)

    const r = await api.activateRule(id)
    const idx = rules.value.findIndex(r => r.id === id)

  }

  async function deactivate(id: number) {
    await api.deactivateRule(id)
    const r = rules.value.find(r => r.id === id)
    if (r) r.active = false
  }

  async function remove(id: number) {
    await api.deleteRule(id)
    rules.value = rules.value.filter(r => r.id !== id)
  }

  return { rules, loading, error, fetchRules, createRule, updateRule, activate, deactivate, remove }

## Task 7: RuleExprBuilder Component

**Files:**
- Create: `web/src/components/RuleExprBuilder.vue`

The component is recursive. The key is `nodeType` as local state — it's the source of truth for which ExprNode variant is rendered. Switching type resets to a fresh default for that variant.

- [ ] **Step 1: Write `RuleExprBuilder.vue`**

```vue
<!-- web/src/components/RuleExprBuilder.vue -->
<script setup lang="ts">
import { ref, watch, computed } from 'vue'
import type { ExprNode } from '../api/rules'
import { useSourcesStore } from '../stores/sources'

type NodeType = 'AND' | 'OR' | 'NOT' | 'src' | 'op'

const props = defineProps<{
  modelValue: ExprNode
  isRoot?: boolean
}>()

const emit = defineEmits<{
  (e: 'update:modelValue', v: ExprNode): void
  (e: 'remove'): void
}>()

const sourcesStore = useSourcesStore()

// Derive current nodeType from the incoming modelValue
function detectType(node: ExprNode): NodeType {
  if ('AND' in node) return 'AND'
  if ('OR' in node)  return 'OR'
  if ('NOT' in node) return 'NOT'
  if ('src' in node) return 'src'
  return 'op'
}

const nodeType = ref<NodeType>(detectType(props.modelValue))

// Local state per variant
const children = ref<ExprNode[]>(
  ('AND' in props.modelValue ? props.modelValue.AND :
   'OR'  in props.modelValue ? props.modelValue.OR  : []) as ExprNode[]
)
const notChild = ref<ExprNode>(
  ('NOT' in props.modelValue ? props.modelValue.NOT : { op: '' }) as ExprNode
)
const srcId = ref<number | null>('src' in props.modelValue ? props.modelValue.src : null)
const opVal = ref<string>('op' in props.modelValue ? props.modelValue.op : '')

// Emit the current node as ExprNode whenever any local state changes
const current = computed<ExprNode>(() => {
  switch (nodeType.value) {
    case 'AND': return { AND: children.value }
    case 'OR':  return { OR:  children.value }
    case 'NOT': return { NOT: notChild.value }
    case 'src': return { src: srcId.value ?? 0 }
    case 'op':  return { op:  opVal.value }
  }
})

watch(current, (v) => emit('update:modelValue', v), { deep: true })

function onTypeChange(t: NodeType) {
  // Reset state when changing type
  nodeType.value = t
  children.value = t === 'AND' || t === 'OR' ? [{ op: '' }] : []
  notChild.value = { op: '' }
  srcId.value    = null
  opVal.value    = ''
}

function addChild() {
  children.value.push({ op: '' })
}

function removeChild(i: number) {
  children.value.splice(i, 1)
}

function updateChild(i: number, v: ExprNode) {
  children.value[i] = v
}

function updateNotChild(v: ExprNode) {
  notChild.value = v
}

const sourceOptions = computed(() =>
  sourcesStore.sources
    .filter(s => s.active)
    .map(s => ({ label: `#${s.id} ${s.name} (${s.operator}${s.value})`, value: s.id }))
)

const typeOptions: { label: string; value: NodeType }[] = [
  { label: 'AND',    value: 'AND' },
  { label: 'OR',     value: 'OR' },
  { label: 'NOT',    value: 'NOT' },
  { label: 'Source', value: 'src' },
  { label: 'Op',     value: 'op' },
]

// Color per type
const typeColor: Record<NodeType, string> = {
  AND: '#f97316', OR: '#2563eb', NOT: '#9333ea', src: '#16a34a', op: '#6b7db3',
}
</script>

<template>
  <div :style="{ borderLeft: `3px solid ${typeColor[nodeType]}`, paddingLeft: '12px', marginBottom: '8px' }">
    <n-space align="center" style="margin-bottom: 6px">
      <n-select
        :value="nodeType"
        :options="typeOptions"
        size="small"
        style="width: 100px"
        @update:value="onTypeChange"
      />

      <!-- src: source dropdown -->
      <n-select
        v-if="nodeType === 'src'"
        v-model:value="srcId"
        :options="sourceOptions"
        size="small"
        style="width: 260px"
        placeholder="Select source"
      />

      <!-- op: free text -->
      <n-input
        v-if="nodeType === 'op'"
        v-model:value="opVal"
        size="small"
        style="width: 200px"
        placeholder="e.g. is:retweet"
      />

      <!-- Add child (AND/OR) -->
      <n-button v-if="nodeType === 'AND' || nodeType === 'OR'" size="small" @click="addChild">
        + Add child
      </n-button>

      <!-- Remove this node (not root) -->
      <n-button v-if="!isRoot" size="small" type="error" @click="emit('remove')">✕</n-button>
    </n-space>

    <!-- AND/OR children -->
    <div v-if="nodeType === 'AND' || nodeType === 'OR'">
      <RuleExprBuilder
        v-for="(child, i) in children"
        :key="i"
        :model-value="child"
        @update:model-value="updateChild(i, $event)"
        @remove="removeChild(i)"
      />
    </div>

    <!-- NOT child -->
    <div v-if="nodeType === 'NOT'">
      <RuleExprBuilder
        :model-value="notChild"
        @update:model-value="updateNotChild"
      />
    </div>
  </div>
</template>
```

  name: string
  event_type: string
  active: boolean
  filter_rule_id: number | null
  filter_model_name: string | null
  filter_model_result: unknown
  pushover_priority: number
  pushover_sound: string | null

  pushover_title: string | null
  pushover_url: string | null
  pushover_url_title: string | null
  pushover_ttl: number | null
  pushover_html: boolean
  pushover_retry: number | null
  pushover_expire: number | null
  message_template: string | null
}

const defaultForm = (): NotificationForm => ({
  name: '', event_type: 'tweet.new', active: true,
  filter_rule_id: null, filter_model_name: null, filter_model_result: null,
  pushover_priority: 0, pushover_html: false,
  pushover_sound: null, pushover_device: null, pushover_title: null,
  pushover_url: null, pushover_url_title: null, pushover_ttl: null,
  pushover_retry: null, pushover_expire: null, message_template: null,
})

  store.fetchNotifications()
  rulesStore.fetchRules()

  editingN.value = null
  form.value = defaultForm()
  showDrawer.value = true
}

  editingN.value = n
  form.value = { ...n }
  showDrawer.value = true

function validateForm(): string | null {
  if (!form.value.name) return 'Name is required'
  if (form.value.pushover_priority === 2) {
    if (!form.value.pushover_retry || form.value.pushover_retry < 30) return 'Retry must be ≥ 30 for emergency priority'
    if (!form.value.pushover_expire || form.value.pushover_expire < 1 || form.value.pushover_expire > 10800)
      return 'Expire must be 1–10800 for emergency priority'
  }
  return null
}

async function onSave() {
  const err = validateForm()
  if (err) { msg.error(err); return }
  saving.value = true
  try {
    if (editingN.value) {
      await store.updateNotification(editingN.value.id, form.value)
    } else {
      await store.createNotification(form.value)
    }
    msg.success('Saved')
    showDrawer.value = false
  } catch (e: any) { msg.error(e.message) }
  finally { saving.value = false }
